1) Fibonnaci

#include <iostream>
using namespace std;

// Recursive function
int fib(int n) {
    if(n <= 1)
        return n;

    return fib(n-1) + fib(n-2);
}

int main() {
    int n;
    cout << "Enter number: ";
    cin >> n;

    cout << "Recursive Fibonacci: ";
    for(int i = 0; i < n; i++) {
        cout << fib(i) << " ";
    }

    cout << "\nNon-Recursive Fibonacci: ";

    int a = 0, b = 1, c;

    for(int i = 0; i < n; i++) {
        cout << a << " ";

        c = a + b;
        a = b;
        b = c;
    }

    return 0;
}

2) Huffman


#include <iostream>
#include <queue>
using namespace std;

struct Node {
    char data;
    int freq;
    Node *left, *right;

    Node(char d, int f) {
        data = d;
        freq = f;
        left = right = NULL;
    }
};

// Min Heap comparator
struct Compare {
    bool operator()(Node* a, Node* b) {
        return a->freq > b->freq;
    }
};

// Print Huffman Codes
void printCodes(Node* root, string code) {
    if (root == NULL)
        return;

    if (root->left == NULL && root->right == NULL) {
        cout << root->data << " : " << code << endl;
    }

    printCodes(root->left, code + "0");
    printCodes(root->right, code + "1");
}

int main() {
    char data[] = {'a', 'b', 'c', 'd', 'e', 'f'};
    int freq[] = {5, 9, 12, 13, 16, 45};

    int n = 6;

    priority_queue<Node*, vector<Node*>, Compare> pq;

    // Insert all nodes into min heap
    for (int i = 0; i < n; i++) {
        pq.push(new Node(data[i], freq[i]));
    }

    // Build Huffman Tree
    while (pq.size() > 1) {

        Node* left = pq.top();
        pq.pop();

        Node* right = pq.top();
        pq.pop();

        Node* top = new Node('$', left->freq + right->freq);

        top->left = left;
        top->right = right;

        pq.push(top);
    }

    // Print Huffman Codes
    Node* root = pq.top();

    cout << "Huffman Codes:\n";
    printCodes(root, "");

    return 0;
}

3) 0-1 Knapsack


#include <bits/stdc++.h>
using namespace std;

int knapsack(vector<vector<int>>& items, int W, int n,
             vector<vector<int>>& dp) {

    // Base condition
    if (n == 0 || W == 0)
        return 0;

    // If already calculated
    if (dp[n][W] != -1)
        return dp[n][W];

    int val = items[n-1][0];
    int wt = items[n-1][1];

    // Include or exclude
    if (wt <= W) {
        int include = val + knapsack(items, W-wt, n-1, dp);
        int exclude = knapsack(items, W, n-1, dp);

        return dp[n][W] = max(include, exclude);
    }
    else {
        return dp[n][W] = knapsack(items, W, n-1, dp);
    }
}

int main() {

    vector<vector<int>> items = {
        {20,2},
        {30,3},
        {50,4},
        {60,5},
        {100,9}
    };

    int W = 10;

    vector<vector<int>> dp(items.size()+1,
                           vector<int>(W+1, -1));

    cout << "Maximum Profit: "
         << knapsack(items, W, items.size(), dp);

    return 0;
}

#include <iostream>
#include <bits/stdc++.h>
using namespace std;

bool isSafe(vector<vector<char>>& board, int row, int col, int n){


    // Check row
    for(int j=0; j<col; j++){
        if(board[row][j]=='Q')
            return false;
    }
    
    
    // Check column
    for(int i=0; i<row; i++){
        if(board[i][col]=='Q')
            return false;
    }

    // Check upper-left diagonal
    for(int i=row, j=col; i>=0 && j>=0; i--, j--){
        if(board[i][j]=='Q')
            return false;
    }

    // Check upper-right diagonal
    for(int i=row, j=col; i>=0 && j<n; i--, j++){
        if(board[i][j]=='Q')
            return false;
    }

    return true;
}

bool nQueens(vector<vector<char>>& board, int row, int n){

    if(row==n)
        return true;

    for(int j=0; j<n; j++){

        if(isSafe(board, row, j, n)){

            board[row][j]='Q';

            if(nQueens(board, row+1, n))
                return true;

            board[row][j]='.';
        }
    }

    return false;
}

int main(){

    int n=4;

    vector<vector<char>> board(n, vector<char>(n,'.'));

    if(nQueens(board, 0, n)){

        for(int i=0; i<n; i++){
            for(int j=0; j<n; j++){
                cout<<board[i][j]<<" ";
            }
            cout<<endl;
        }
    }
    return 0;
}

5) Quick sort


#include <bits/stdc++.h>
using namespace std;

int partition(vector<int>& a, int low, int high) {

    // Select random pivot
    int r = low + rand() % (high - low + 1);

    swap(a[r], a[high]);

    int pivot = a[high];
    int i = low - 1;

    for(int j = low; j < high; j++) {

        if(a[j] <= pivot) {
            i++;
            swap(a[i], a[j]);
        }
    }

    swap(a[i+1], a[high]);

    return i+1;
}

void quickSort(vector<int>& a, int low, int high) {

    if(low < high) {

        int pi = partition(a, low, high);

        quickSort(a, low, pi-1);
        quickSort(a, pi+1, high);
    }
}

int main() {

    srand(time(0));

    vector<int> a = {40, 20, 10, 80, 60, 50};

    quickSort(a, 0, a.size()-1);

    cout << "Sorted array: ";

    for(int x : a)
        cout << x << " ";

    return 0;
}

#include <bits/stdc++.h>
using namespace std;

int partition(vector<int>& arr, int low, int high) {

    int pivot = arr[high];
    int idx = low - 1;

    for(int j = low; j < high; j++) {

        if(arr[j] <= pivot) {
            idx++;
            swap(arr[j], arr[idx]);
        }
    }

    idx++;
    swap(arr[high], arr[idx]);

    return idx;
}

int randomPartition(vector<int>& arr, int low, int high) {

    int rnd = low + rand() % (high - low + 1);

    swap(arr[rnd], arr[high]);

    return partition(arr, low, high);
}

void QS(vector<int>& arr, int low, int high) {

    if(low < high) {

        int pvt = randomPartition(arr, low, high);

        QS(arr, low, pvt - 1);
        QS(arr, pvt + 1, high);
    }
}

int main() {

    vector<int> arr = {8, 3, 1, 7, 0, 10, 2};

    QS(arr, 0, arr.size() - 1);

    for(int x : arr)
        cout << x << " ";

    return 0;
}
